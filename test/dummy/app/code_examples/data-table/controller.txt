SORTS = {
  "customer" => "invoices.customer",
  "email" => "invoices.email",
  "status" => "invoices.status",
  "amount" => "invoices.amount"
}.freeze

def index
  scope = Invoice.all

  if params[:q].present?
    query = "%#{params[:q]}%"
    scope = scope.where("customer ILIKE :query OR email ILIKE :query", query: query)
  end

  scope = scope.where(status: params[:status]) if params[:status].present?

  if SORTS.key?(params[:sort]) && %w[asc desc].include?(params[:dir])
    scope = scope.order(SORTS.fetch(params[:sort]) => params[:dir])
  end

  @total_count = scope.count
  @pagy, @invoices = pagy(scope)
end
